# TEST 24 - Tier Yükseltme
# Müşteri önce REGULAR, tier GOLD'a yükseltilir.
# Yeni siparişlerde GOLD indirimi geçerli.

ADD_CATEGORY GEN Genel 0
ADD_PRODUCT P001 Urun GEN 1000.00 50
ADD_CUSTOMER C001 Yukselen_Musteri REGULAR

# REGULAR ile sipariş (indirim %0)
CREATE_ORDER O001 C001

FINALIZE_ORDER O001
ORDER_DETAIL O001
# Beklenen: 1000.00

UPGRADE_TIER C001 GOLD

# GOLD ile sipariş (indirim %10)
CREATE_ORDER O002 C001
ADD_ITEM O002 P001 1

# Beklenen: 1000 * 0.90 = 900.00
